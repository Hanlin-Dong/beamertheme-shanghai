\begin{frame}{1.1 地理与人口}
  \PlainHeading{长江三角洲的龙头}
  \begin{itemize}
    \item 地处长江入海口，东临东海，黄浦江穿城而过
    \item 全市面积约 \EmphNum{6340.5} 平方千米，辖 \EmphNum{16} 个区
    \item 常住人口约 \EmphNum{2487} 万（2020 年第七次全国人口普查）
  \end{itemize}

  \PlainHeading{平原水乡}
  \begin{itemize}
    \item 平均海拔 \EmphNum{4} 米左右，大金山岛为全市最高点
    \item 淀山湖、崇明岛构成西部的湖群与沙岛景观
  \end{itemize}
\end{frame}

% ---------- 强调框页：重点数据用 block ----------
\begin{frame}{1.2 经济与交通}
  \begin{block}{经济总量}
    2024 年地区生产总值约 \EmphNum{5.39} 万亿元，是我国最大的经济中心城市之一；
    港口集装箱吞吐量连续多年居世界第一。
  \end{block}

  \PlainHeading{轨道交通}
  \begin{itemize}
    \item 运营里程超过 \EmphNum{800} 千米，线网规模居世界前列
    \item 日常工作日客流逾 \EmphNum{1000} 万人次
  \end{itemize}

  \PlainHeading{航空枢纽}
  浦东、虹桥两大国际机场，通航全球 \EmphNum{300} 多个城市。
\end{frame}

\begin{frame}{2.1 从渔村到开埠}
  \PlainHeading{古代：青龙镇与上海滩}
  \begin{itemize}
    \item 唐宋时期青龙镇已是江南贸易港口
    \item 元至元二十九年（\EmphNum{1292}）设上海县
  \end{itemize}

  \PlainHeading{近代：开埠与工商都会}
  \begin{itemize}
    \item \EmphNum{1843} 年开埠，租界时期形成外滩建筑群
    \item 民族工业、报馆、大学在此聚集，孕育现代都市文明
  \end{itemize}

  \PlainHeading{现代：改革开放前沿}
  \begin{itemize}
    \item \EmphNum{1990} 年浦东开发开放，陆家嘴从阡陌农田变身金融城
  \end{itemize}
\end{frame}

\begin{frame}{2.2 海派文化}
  \begin{block}{城市精神}
    海纳百川，追求卓越；开明睿智，大气谦和。
  \end{block}

  \PlainHeading{文化切片}
  \begin{itemize}
    \item 石库门：中西合璧的里弄住宅，中共一大会址所在（\EmphNum{1921}）
    \item 沪剧、评弹、滑稽戏：从市井生长出来的舞台艺术
    \item 梧桐区：衡复风貌区的花园洋房与咖啡馆
  \end{itemize}
\end{frame}
